(defn assert-exit-status [result expected-status]

2. CI

2.1. Execution times

  • Install Clojure & BB, no Emacs: 11s.
# Please note that this is a duplicate. The source is in chapters/development.org.
name: CI
on: [push, pull_request]

jobs:
  build:
    runs-on: ubuntu-latest

    strategy:
      matrix:
        os: [ubuntu-latest]
        java: [11]

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up JDK ${{ matrix.java }}
      uses: actions/setup-java@v2
      with:
        distribution: temurin
        java-version: ${{ matrix.java }}

    - name: Install Clojure CLI
      run: |
        curl -O https://download.clojure.org/install/linux-install-1.10.3.998.sh
        chmod +x linux-install-1.10.3.998.sh
        sudo ./linux-install-1.10.3.998.sh

    - name: Install Babashka
      run: |
        wget https://github.com/babashka/babashka/releases/download/v1.3.191/babashka-1.3.191-linux-amd64-static.tar.gz -O babashka-release.tgz
        tar -xzf babashka-release.tgz
        sudo mv bb /usr/local/bin/bb

    - name: Tangle Org files
      run: |
        for file in chapters/*.org; do
          echo TODO
        done

    - name: Run tests in JVM
      run: clojure -M:test

    - name: Run tests in Babashka
      run: bb -cp "$(clojure -Spath -Mtest)" run-tests.bb
