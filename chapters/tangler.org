* Tangler
Replacement for the tangling functionality of org-babel. Apart from tangling and NoWeb is fully org-mode compatible and ...
Note that you can use Tangler in conjunction with org-babel, especially if you want NoWeb or code execution within the buffer.

- Whitespace indent reset.
- Shebang handling (chmod +x)
- Args validator: warn on unrecognised opts...also validate <<NOWEB>>.
  - Probably not built-in, but rather use middlewares and have the middlewares not only call up, but to remove arguments that it processed?
- Publish first version and install it in the CI script.

#+begin_src clojure :tangle ../src/jakub_stastny/tangler/core.clj :mkdirp yes :ns jakub-stastny.tangler.core
  (ns jakub-stastny.tangler.core
    (:require [clojure.string :as str]
              [clojure.java.io :as io]
              [clojure.string :as str]
              [jakub-stastny.org-mode.parser :as parser]
              [jakub-stastny.tangler.utils :refer [ensure-dir-empty]]
              [jakub-stastny.tangler.cli :refer [parse-args]]))

  (defn ns-to-path [ns-name ext]
    (str "src/" (str/replace ns-name "." "/") "." ext))

  ;; TODO: Handle with defmulti maybe?
  (defn relative-file-path [lang {:keys [path ext ns] :as opts}]
    (cond
      ;; If path is used, no extension is added automatically (but :ext is respected).
      ;; :path "bin/script" :ext "bb"                  -> bin/script.bb
      (and path ext) (str path "." ext)

      ;; :path "bin/script"                            -> bin/script
      path (str path)

      ;; :ns jakub-stastny.org-mode.parser             -> src/jakub_stastny/org_mode/parser.clj
      ;; :ns jakub-stastny.org-mode.parser :ext "cljc" -> src/jakub_stastny/org_mode/parser.cljc

      ;; TODO: :ns points to src by default. It should be configurable to src: ns-name, test: ns-name-test etc.
      ;; If it ends with -test, it's in: "test"
      ;; Or maybe use :ns ... in: "test"
      ns (ns-to-path (str ns (or ext "clj")))

      ;; TODO: Which chapter from, number of code group?
      true (throw (ex-info "Cannot construct relative-file-path" {:lang lang :opts opts}))))

  (defn format-block [{:keys [lang lines opts]}]
    ;; Skip ET tasks and :export no/false.
    ;; TODO: export is used by ox- packages, name it differently.
    (when-not (or (contains? #{'no false} (:export opts)) (or (:task opts) (:doc opts)))
      ;; TODO: Add shebang (from opts), where should preamble be added? It comes from configuration.
      {:path (relative-file-path lang opts) :code (str/join "\n" lines)}))

  (defn format-blocks [blocks]
    (reduce (fn [acc block]
              (let [result (format-block block)]
                (if (:path result) (conj acc result) acc)))
            [] blocks))

  (defn write-block [output-dir {:keys [path code]}]
    (let [full-path (str output-dir "/" path)]
      (println (str "~ " path))
      (spit full-path code :append true)))

  (defn write-blocks [output-dir blocks]
    (doseq [block blocks]
      (write-block output-dir block)))

  (defn tangle [org-file output-dir]
    (ensure-dir-empty output-dir)
    (let [lines (str/split-lines (slurp org-file))
          examples (parser/parse-examples lines)
          code-blobs (format-blocks examples)]
      (write-blocks output-dir code-blobs)))
#+end_src

#+begin_src clojure :tangle ../src/jakub_stastny/tangler/utils.clj :mkdirp yes :ns jakub-stastny.tangler.utils
  (ns jakub-stastny.tangler.utils
    (:require [clojure.string :as str]
              [clojure.java.io :as io]
              [babashka.fs :as fs]))

  (defn ensure-dir-empty [path]
    (fs/delete-tree path)
    (fs/create-dirs path))
#+end_src

#+begin_src clojure :tangle ../src/jakub_stastny/tangler/cli.clj :mkdirp yes :ns jakub-stastny.tangler.cli
  (ns jakub-stastny.tangler.cli
    (:require [babashka.cli :as cli]))

  (defn parse-args [args]
    (cli/parse-opts args
                    [["-f" "--file FILE" "Org mode file" :default "example.org"]
                     ["-o" "--output-dir DIR" "Output directory" :default "."]]))
#+end_src

#+begin_src clojure :tangle ../src/jakub_stastny/tangler/main.clj :mkdirp yes :ns jakub-stastny.tangler.main
  (ns jakub-stastny.tangler.main
    (:require [jakub-stastny.tangler.core :refer [tangle]]
              [jakub-stastny.tangler.cli :refer [parse-args]]))

  (defn -main [& args]
    (let [{:keys [file output-dir]} (parse-args args)]
      (tangle file output-dir)))

  ;; Check if the script is being run with arguments and call -main if so.
  (when (= *ns* 'jakub-stastny.tangler.main)
    (apply -main *command-line-args*))
#+end_src
