*** Clojure configuration ~deps.edn~
#+begin_src clojure :tangle ../deps.edn
  {:paths ["src"]
   :deps {org.clojure/clojure {:mvn/version "1.11.3"}}
   :aliases
   {:build
    {:extra-deps {io.github.clojure/tools.build {:mvn/version "0.10.4"}}
     :ns-default build}

    :tangler
    {:extra-deps {org.babashka/cli {:mvn/version "0.8.59"}}}

    :test
    {:extra-paths ["test"]
     :extra-deps {org.clojure/test.check {:mvn/version "1.1.1"}
                  io.github.cognitect-labs/test-runner {:git/tag "v0.5.1" :git/sha "dfb30dd"}
                  babashka/process {:mvn/version "0.5.22"}}
     :main-opts ["-m" "cognitect.test-runner"]}}}
#+end_src

*** Babashka configuration ~bb.edn~
- Now you can run ~bb run tangle~ and ~bb run test~.

#+begin_src clojure :tangle ../bb.edn
  {:paths ["src" "test"]
   :deps {org.babashka/cli {:mvn/version "0.8.59"}}
   :tasks
   {:tangle
    {:doc "Run the tangler script"
     :task (shell {:dir "src" :cmd "bb -m jakub-stastny.tangler.main --file chapters/build.org --output-dir xxx"})}

    :test
    {:doc "Run the tests"
     :task (shell {:dir "test" :cmd "bb -m cognitect.test-runner"})}}}
#+end_src

#+begin_src clojure :tangle ../build.clj
  (ns build
    (:require [clojure.tools.build.api :as b]))

  (def lib 'jakub-stastny/et)
  (def class-dir "target/classes")
  (def basis (b/create-basis {:project "deps.edn"}))
  (def uber-file (format "target/%s.jar" (name lib)))

  (defn clean [_] (b/delete {:path "target"}))

  (defn uber [_]
    (clean nil)
    (b/copy-dir {:src-dirs ["src"] :target-dir class-dir})
    (b/compile-clj {:basis basis :src-dirs ["src"] :class-dir class-dir})
    (b/uber {:class-dir class-dir :uber-file uber-file
             :basis basis :main 'jakub-stastny.et.runner}))
#+end_src
